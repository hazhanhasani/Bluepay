نمونه Python برای سایت

import httpx

API_KEY = "gw_your_private_key"
BASE_URL = "{{ base_url }}"

payload = {
    "amount_toman": 200000,
    "order_id": "ORDER-1001",
    "description": "اشتراک یک ماهه",
    "fee_mode": "split",
}

response = httpx.post(
    f"{BASE_URL}/api/v1/invoices",
    headers={"X-API-Key": API_KEY},
    json=payload,
    timeout=15,
)
response.raise_for_status()
invoice = response.json()
print(invoice["payment_url"])

اتصال ربات تلگرام

نمونه ساده با aiogram و httpx

from aiogram import Router
from aiogram.types import Message, InlineKeyboardMarkup, InlineKeyboardButton
import httpx

router = Router()
API_KEY = "gw_your_private_key"
BASE_URL = "{{ base_url }}"

@router.message(lambda message: message.text == "خرید")
async def buy(message: Message):
    async with httpx.AsyncClient(timeout=15) as client:
        response = await client.post(
            f"{BASE_URL}/api/v1/invoices",
            headers={"X-API-Key": API_KEY},
            json={
                "amount_toman": 200000,
                "order_id": f"TG-{message.from_user.id}-{message.message_id}",
                "description": "خرید از ربات"
            },
        )
    response.raise_for_status()
    invoice = response.json()
    keyboard = InlineKeyboardMarkup(inline_keyboard=[[
        InlineKeyboardButton(text="پرداخت", url=invoice["payment_url"])
    ]])
    await message.answer("برای پرداخت روی دکمه بزن:", reply_markup=keyboard)

Callback نتیجه پرداخت

وبهوک خروجی اختصاصی هر پذیرنده

پس از تأیید، یک درخواست POST با رویداد invoice.paid به آدرس ثبت‌شده ارسال می‌شود. در صورت خطا، ارسال تا سه مرتبه تکرار می‌شود.

X-Gateway-SignatureX-Gateway-EventX-Gateway-DeliveryX-Gateway-Timestamp
{
  "event": "invoice.paid",
  "payment_id": "PAYMENT_TOKEN",
  "order_id": "ORDER-1001",
  "status": "paid",
  "base_amount_rial": 2000000,
  "fee_amount_rial": 20000,
  "customer_fee_rial": 10000,
  "unique_amount_rial": 3700,
  "payable_amount_rial": 2013700,
  "reference_number": "998877",
  "paid_at": "2026-07-31T18:12:10+00:00"
}

اعتبارسنجی امضا در Python

import hashlib, hmac, json

def verify_webhook(raw_body: bytes, received_signature: str, secret: str) -> bool:
    payload = json.loads(raw_body)
    canonical = json.dumps(
        payload,
        ensure_ascii=False,
        separators=(",", ":"),
        sort_keys=True,
    ).encode("utf-8")
    expected = hmac.new(secret.encode(), canonical, hashlib.sha256).hexdigest()
    return hmac.compare_digest(expected, received_signature)

اعتبارسنجی امضا در PHP

<?php
$secret = 'SECRET_FROM_BOT';
$raw = file_get_contents('php://input');
$payload = json_decode($raw, true);
ksort($payload);
$canonical = json_encode($payload, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES);
$expected = hash_hmac('sha256', $canonical, $secret);
$received = $_SERVER['HTTP_X_GATEWAY_SIGNATURE'] ?? '';
if (!hash_equals($expected, $received)) {
    http_response_code(401);
    exit('invalid signature');
}
http_response_code(200);
echo 'ok';
برای جلوگیری از پردازش تکراری، مقدار X-Gateway-Delivery یا payment_id را در دیتابیس خود ذخیره و یکتا کن.

وبهوک اختصاصی پیامک بانکی

هر پذیرنده یک URL مستقل و محرمانه دارد

POST{{ sms_webhook_url }}

ساخت Body در SMS Forwarder

{"device_id":"phone-1","sender":"","message":""}
مهم: عبارت‌های {Incoming Number} و {Message Body} را تایپ یا از سایت کپی نکن. این دو باید به‌صورت فیلد پویا از بخش Message Template یا فهرست متغیرهای خود SMS Forwarder درج شوند.
۱در برنامه SMS Forwarder یک Filter یا Rule برای پیامک‌های بانکی بساز.
۲مقصد را Webhook URL و روش ارسال را POST انتخاب کن.
۳Webhook اختصاصی بالا را بدون تغییر وارد کن.
۴در قسمت Body گزینه JSON را انتخاب و اسکلت JSON بالا را وارد کن.
۵داخل مقدار sender، فیلد پویا Incoming Number را از خود برنامه درج کن.
۶داخل مقدار message، فیلد پویا Message Body را از خود برنامه درج کن.
۷مقدار phone-1 نام این گوشی است و برای گوشی‌های دیگر باید متفاوت باشد.
۸اگر فیلدها به‌صورت متن ساده ارسال شوند، پاسخ SMS_TEMPLATE_NOT_RESOLVED دریافت می‌کنی و پیامک پردازش نمی‌شود.
۹نام بانک را دستی وارد نکن. سیستم بانک را از فرستنده و متن واقعی پیامک تشخیص می‌دهد.
نمونه JSON برای برنامه‌های دیگر 
{
  "sender": "VARIABLE_FOR_SMS_SENDER",
  "message": "VARIABLE_FOR_SMS_BODY",
  "device_id": "phone-1"
}

نام متغیرها در برنامه‌های مختلف فرق می‌کند. فیلد bank_code اختیاری است و برای اتصال عادی لازم نیست.

URL اختصاصی پیامک نقش کلید امنیتی دارد. آن را در کانال، گروه یا کد عمومی منتشر نکن.
